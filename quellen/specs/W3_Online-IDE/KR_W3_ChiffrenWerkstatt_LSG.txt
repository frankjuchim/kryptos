class Hilfe {

    // normalisiere(text: Zeichenkette): Zeichenkette
    // Grossbuchstaben bleiben, Kleinbuchstaben werden gross, ae/oe/ue/ss
    // fuer Umlaute und Eszett, alle anderen Zeichen fallen weg.
    // Beispiel: "Grüße an Bob!" -> "GRUESSEANBOB"
    public static String normalisiere(String text) {
        String s = "";
        for (int i = 0; i < text.length(); i++) {
            char c = text.charAt(i);
            int a = (int) c;
            if (a >= 65 && a <= 90) {
                s = s + c;
            } else if (a >= 97 && a <= 122) {
                s = s + (char) (a - 32);
            } else if (a == 228 || a == 196) {
                s = s + "AE";
            } else if (a == 246 || a == 214) {
                s = s + "OE";
            } else if (a == 252 || a == 220) {
                s = s + "UE";
            } else if (a == 223) {
                s = s + "SS";
            }
        }
        return s;
    }

    // mod(a: Ganzzahl, m: Ganzzahl): Ganzzahl
    // kleinster nicht negativer Rest (wie mod in der Abitur-Anlage),
    // z. B. mod(-3, 26) = 23. Achtung: In Java ergibt -3 % 26 = -3!
    public static int mod(int a, int m) {
        int r = a % m;
        if (r < 0) {
            r = r + m;
        }
        return r;
    }

    // wert(c: Zeichen): Ganzzahl
    // Buchstabenwert mit A = 0 ... Z = 25 (ASCII-Wert minus 65).
    // ACHTUNG Falle P1: Auf Papier gilt A = 1 ... Z = 26!
    public static int wert(char c) {
        return (int) c - 65;
    }

    // zeichen(w: Ganzzahl): Zeichen
    // Buchstabe zum Wert w (A = 0); w wird vorher modulo 26 genommen.
    public static char zeichen(int w) {
        return (char) (mod(w, 26) + 65);
    }

    // hex2(n: Ganzzahl): Zeichenkette
    // Zahl von 0 bis 255 als zwei Hexadezimalziffern, z. B. 7 -> "07",
    // 26 -> "1A", 255 -> "FF".
    public static String hex2(int n) {
        String ziffern = "0123456789ABCDEF";
        return "" + ziffern.charAt(n / 16) + ziffern.charAt(n % 16);
    }

    // hexWert(c: Zeichen): Ganzzahl
    // Wert einer Hexadezimalziffer '0'...'9', 'A'...'F' (0 bis 15).
    public static int hexWert(char c) {
        int a = (int) c;
        if (a >= 48 && a <= 57) {
            return a - 48;
        }
        return a - 55;
    }
}

// ==================== Level 3.3 – Die Bausteine =======================

class Bausteine {

    // ===================== Stufe BASIS =====================

    // caesar(text: Zeichenkette, k: Ganzzahl): Zeichenkette
    public static String caesar(String text, int k) {
        String erg = "";
        for (int i = 0; i < text.length(); i++) {
            erg = erg + Hilfe.zeichen(Hilfe.wert(text.charAt(i)) + k);
        }
        return erg;
    }

    // caesarZurueck(geheimtext: Zeichenkette, k: Ganzzahl): Zeichenkette
    public static String caesarZurueck(String geheimtext, int k) {
        return caesar(geheimtext, 26 - Hilfe.mod(k, 26));
    }

    // umkehren(text: Zeichenkette): Zeichenkette
    public static String umkehren(String text) {
        String erg = "";
        for (int i = text.length() - 1; i >= 0; i--) {
            erg = erg + text.charAt(i);
        }
        return erg;
    }


    // ================== Stufe ERWEITERUNG ==================

    // vigenere(text: Zeichenkette, wort: Zeichenkette): Zeichenkette
    public static String vigenere(String text, String wort) {
        String erg = "";
        for (int i = 0; i < text.length(); i++) {
            int k = Hilfe.wert(wort.charAt(i % wort.length()));
            erg = erg + Hilfe.zeichen(Hilfe.wert(text.charAt(i)) + k);
        }
        return erg;
    }

    // vigenereZurueck(geheimtext: Zeichenkette, wort: Zeichenkette): Zeichenkette
    public static String vigenereZurueck(String geheimtext, String wort) {
        String erg = "";
        for (int i = 0; i < geheimtext.length(); i++) {
            int k = Hilfe.wert(wort.charAt(i % wort.length()));
            erg = erg + Hilfe.zeichen(Hilfe.wert(geheimtext.charAt(i)) - k);
        }
        return erg;
    }

    // zeile(i: Ganzzahl, h: Ganzzahl): Ganzzahl   (Hilfsmethode)
    // In welcher Zeile (0 bis h - 1) des Gartenzauns steht Zeichen i?
    public static int zeile(int i, int h) {
        int periode = 2 * (h - 1);
        int m = i % periode;
        if (m < h) {
            return m;
        }
        return periode - m;
    }

    // gartenzaun(text: Zeichenkette, h: Ganzzahl): Zeichenkette
    public static String gartenzaun(String text, int h) {
        if (h < 2) {
            return text;
        }
        String erg = "";
        for (int z = 0; z < h; z++) {
            for (int i = 0; i < text.length(); i++) {
                if (zeile(i, h) == z) {
                    erg = erg + text.charAt(i);
                }
            }
        }
        return erg;
    }

    // gartenzaunZurueck(geheimtext: Zeichenkette, h: Ganzzahl): Zeichenkette
    public static String gartenzaunZurueck(String geheimtext, int h) {
        if (h < 2) {
            return geheimtext;
        }
        char[] klar = new char[geheimtext.length()];
        int pos = 0;
        for (int z = 0; z < h; z++) {
            for (int i = 0; i < geheimtext.length(); i++) {
                if (zeile(i, h) == z) {
                    klar[i] = geheimtext.charAt(pos);
                    pos++;
                }
            }
        }
        String erg = "";
        for (int i = 0; i < klar.length; i++) {
            erg = erg + klar[i];
        }
        return erg;
    }


    // ==================== Stufe EXPERTE ====================

    // rang(wort: Zeichenkette, j: Ganzzahl): Ganzzahl   (Hilfsmethode)
    public static int rang(String wort, int j) {
        int r = 0;
        for (int i = 0; i < wort.length(); i++) {
            char a = wort.charAt(i);
            char b = wort.charAt(j);
            if (a < b || (a == b && i < j)) {
                r++;
            }
        }
        return r;
    }

    // spalteMitRang(wort: Zeichenkette, nr: Ganzzahl): Ganzzahl   (Hilfsmethode)
    public static int spalteMitRang(String wort, int nr) {
        for (int j = 0; j < wort.length(); j++) {
            if (rang(wort, j) == nr) {
                return j;
            }
        }
        return -1;
    }

    // spaltentransposition(text: Zeichenkette, wort: Zeichenkette): Zeichenkette
    public static String spaltentransposition(String text, String wort) {
        int n = wort.length();
        String erg = "";
        for (int nr = 0; nr < n; nr++) {
            int spalte = spalteMitRang(wort, nr);
            for (int i = spalte; i < text.length(); i = i + n) {
                erg = erg + text.charAt(i);
            }
        }
        return erg;
    }

    // spaltentranspositionZurueck(geheimtext: Zeichenkette, wort: Zeichenkette): Zeichenkette
    public static String spaltentranspositionZurueck(String geheimtext, String wort) {
        int n = wort.length();
        char[] klar = new char[geheimtext.length()];
        int pos = 0;
        for (int nr = 0; nr < n; nr++) {
            int spalte = spalteMitRang(wort, nr);
            for (int i = spalte; i < geheimtext.length(); i = i + n) {
                klar[i] = geheimtext.charAt(pos);
                pos++;
            }
        }
        String erg = "";
        for (int i = 0; i < klar.length; i++) {
            erg = erg + klar[i];
        }
        return erg;
    }

    // xorHex(text: Zeichenkette, wort: Zeichenkette): Zeichenkette
    public static String xorHex(String text, String wort) {
        String erg = "";
        for (int i = 0; i < text.length(); i++) {
            int b = (int) text.charAt(i) ^ (int) wort.charAt(i % wort.length());
            erg = erg + Hilfe.hex2(b);
        }
        return erg;
    }

    // xorHexZurueck(hex: Zeichenkette, wort: Zeichenkette): Zeichenkette
    public static String xorHexZurueck(String hex, String wort) {
        String erg = "";
        for (int i = 0; i < hex.length() / 2; i++) {
            int b = Hilfe.hexWert(hex.charAt(2 * i)) * 16 + Hilfe.hexWert(hex.charAt(2 * i + 1));
            erg = erg + (char) (b ^ (int) wort.charAt(i % wort.length()));
        }
        return erg;
    }
}


// ================== Level 3.3 – Deine eigene Chiffre ==================
//  Beispielverfahren FUNKGITTER (Musterloesung):
//    Schluessel: ein Schluesselwort aus Grossbuchstaben, mind. 2 Zeichen
//    Stufe 1: Vigenere mit dem Schluesselwort
//    Stufe 2: Spaltentransposition mit demselben Schluesselwort
//    Entschluesseln: Stufe 2 zurueck, dann Stufe 1 zurueck (P3!)

class MeineChiffre {

    public static final String NAME = "FUNKGITTER";
    public static final String BEISPIEL_KLARTEXT = "Wetterbericht fuer die Nordsee";
    public static final String BEISPIEL_SCHLUESSEL = "FUNKE";
    public static final String BEISPIEL_GEHEIMTEXT = "IMYIWBWHJSJDBPSNGRGQEYVBLIY";
    public static final String TESTSCHLUESSEL_1 = "FUNKE";
    public static final String TESTSCHLUESSEL_2 = "RADAR";

    // verschluesseln(klartext: Zeichenkette, schluessel: Zeichenkette): Zeichenkette
    public static String verschluesseln(String klartext, String schluessel) {
        String t = Hilfe.normalisiere(klartext);
        String stufe1 = Bausteine.vigenere(t, schluessel);
        return Bausteine.spaltentransposition(stufe1, schluessel);
    }

    // entschluesseln(geheimtext: Zeichenkette, schluessel: Zeichenkette): Zeichenkette
    public static String entschluesseln(String geheimtext, String schluessel) {
        String stufe1 = Bausteine.spaltentranspositionZurueck(geheimtext, schluessel);
        return Bausteine.vigenereZurueck(stufe1, schluessel);
    }
}

class L33BausteineTest {

    @Test
    public void basis_caesar() {
        assertEquals("IXQNVWLOOH", Bausteine.caesar("FUNKSTILLE", 3), "caesar(\"FUNKSTILLE\", 3) falsch");
        assertEquals("ABC", Bausteine.caesar("ABC", 0), "Mit k = 0 bleibt der Text gleich (schwacher Schluessel!)");
    }

    @Test
    public void basis_caesar_ueberlauf() {
        assertEquals("BCD", Bausteine.caesar("XYZ", 4), "Ueberlauf hinter Z: X + 4 = B - benutze Hilfe.zeichen (rechnet modulo 26)");
        assertEquals("YDAQZ", Bausteine.caesar("ZEBRA", 25), "caesar(\"ZEBRA\", 25) falsch - Verschiebung um 25 = um 1 zurueck");
    }

    @Test
    public void basis_caesarZurueck() {
        assertEquals("FUNKSTILLE", Bausteine.caesarZurueck("IXQNVWLOOH", 3), "caesarZurueck(\"IXQNVWLOOH\", 3) falsch - Falle P3: zurueck heisst -k");
        assertEquals("XYZ", Bausteine.caesarZurueck("BCD", 4), "caesarZurueck(\"BCD\", 4) falsch - negative Werte? Hilfe.mod bzw. Hilfe.zeichen benutzen (Falle P2)");
    }

    @Test
    public void basis_umkehren() {
        assertEquals("DNIWDRON", Bausteine.umkehren("NORDWIND"), "umkehren(\"NORDWIND\") falsch");
        assertEquals("A", Bausteine.umkehren("A"), "Ein einzelnes Zeichen bleibt gleich");
        assertEquals("", Bausteine.umkehren(""), "Leerer Text bleibt leer");
    }

    @Test
    public void erweiterung_vigenere() {
        assertEquals("BYGDJLOOWCPRY", Bausteine.vigenere("WETTERBERICHT", "FUNK"), "vigenere(\"WETTERBERICHT\", \"FUNK\") falsch - Schluesselbuchstabe A bedeutet Verschiebung 0");
        assertEquals("XIEGPHEOE", Bausteine.vigenere("MORGENROT", "LUNA"), "vigenere(\"MORGENROT\", \"LUNA\") falsch - Schluesselwort wiederholen: wort.charAt(i % wort.length())");
        assertEquals("AAA", Bausteine.vigenere("ZZZ", "B"), "vigenere(\"ZZZ\", \"B\") = \"AAA\" (Ueberlauf)");
    }

    @Test
    public void erweiterung_vigenereZurueck() {
        assertEquals("WETTERBERICHT", Bausteine.vigenereZurueck("BYGDJLOOWCPRY", "FUNK"), "vigenereZurueck falsch - Falle P3: Schluesselwert abziehen");
        assertEquals("ZZZ", Bausteine.vigenereZurueck("AAA", "B"), "vigenereZurueck(\"AAA\", \"B\") = \"ZZZ\" - negative Werte modulo 26 (Falle P2)");
    }

    @Test
    public void erweiterung_gartenzaun() {
        assertEquals("FSERENCRIERHB", Bausteine.gartenzaun("FERNSCHREIBER", 3), "gartenzaun(\"FERNSCHREIBER\", 3) falsch - Zickzack: Zeilen 0,1,2,1,0,1,2,...");
        assertEquals("ACEGBDFH", Bausteine.gartenzaun("ABCDEFGH", 2), "gartenzaun(\"ABCDEFGH\", 2) falsch");
        assertEquals("AGBFHCEIDJ", Bausteine.gartenzaun("ABCDEFGHIJ", 4), "gartenzaun(\"ABCDEFGHIJ\", 4) falsch");
    }

    @Test
    public void erweiterung_gartenzaunZurueck() {
        assertEquals("FERNSCHREIBER", Bausteine.gartenzaunZurueck("FSERENCRIERHB", 3), "gartenzaunZurueck(\"FSERENCRIERHB\", 3) falsch");
        assertEquals("ABCDEFGHIJ", Bausteine.gartenzaunZurueck("AGBFHCEIDJ", 4), "gartenzaunZurueck(\"AGBFHCEIDJ\", 4) falsch");
    }

    @Test
    public void experte_spaltentransposition() {
        assertEquals("RCEEESRBFNHIR", Bausteine.spaltentransposition("FERNSCHREIBER", "ZUG"), "Spalten in alphabetischer Reihenfolge der Schluesselbuchstaben auslesen (G, U, Z) - letzte Zeile bleibt unvollstaendig, kein X");
        assertEquals("NRTITHHDSCCNNAIEE", Bausteine.spaltentransposition("NACHRICHTENDIENST", "FUNK"), "spaltentransposition(\"NACHRICHTENDIENST\", \"FUNK\") falsch");
    }

    @Test
    public void experte_spaltentransposition_gleicheBuchstaben() {
        assertEquals("LSICREOTFHEN", Bausteine.spaltentransposition("LOCHSTREIFEN", "BEBE"), "Gleiche Schluesselbuchstaben werden von links nach rechts nummeriert (Falle P4)");
    }

    @Test
    public void experte_spaltentranspositionZurueck() {
        assertEquals("FERNSCHREIBER", Bausteine.spaltentranspositionZurueck("RCEEESRBFNHIR", "ZUG"), "Ungleich lange Spalten beachten: die ersten (Laenge % n) Spalten sind eine Zeile laenger (Falle P4)");
        assertEquals("NACHRICHTENDIENST", Bausteine.spaltentranspositionZurueck("NRTITHHDSCCNNAIEE", "FUNK"), "spaltentranspositionZurueck mit FUNK falsch");
        assertEquals("LOCHSTREIFEN", Bausteine.spaltentranspositionZurueck("LSICREOTFHEN", "BEBE"), "spaltentranspositionZurueck mit BEBE falsch");
    }

    @Test
    public void experte_xorHex() {
        assertEquals("07170F09", Bausteine.xorHex("FUNK", "AB"), "xorHex(\"FUNK\", \"AB\") falsch - ASCII-Werte mit ^ verknuepfen, je Byte zwei Hexziffern (Hilfe.hex2)");
        assertEquals("0504190F", Bausteine.xorHex("NORD", "K"), "xorHex(\"NORD\", \"K\") falsch");
    }

    @Test
    public void experte_xorHexZurueck() {
        assertEquals("FUNK", Bausteine.xorHexZurueck("07170F09", "AB"), "xorHexZurueck falsch - XOR ist selbstinvers: noch einmal mit demselben Schluessel");
        assertEquals("NORD", Bausteine.xorHexZurueck("0504190F", "K"), "xorHexZurueck(\"0504190F\", \"K\") falsch");
    }
}


class L33MeineChiffreTest {

    // nurBuchstabenOderHex(s: Zeichenkette): Wahrheitswert
    public static boolean nurBuchstabenOderHex(String s) {
        for (int i = 0; i < s.length(); i++) {
            int a = (int) s.charAt(i);
            boolean buchstabe = a >= 65 && a <= 90;
            boolean ziffer = a >= 48 && a <= 57;
            if (!buchstabe && !ziffer) {
                return false;
            }
        }
        return true;
    }

    @Test
    public void beispielwert_ausDemBaukasten() {
        assertTrue(MeineChiffre.BEISPIEL_GEHEIMTEXT.length() > 0, "Trage in MeineChiffre das Beispiel aus dem Baukasten ein (BEISPIEL_KLARTEXT, BEISPIEL_SCHLUESSEL, BEISPIEL_GEHEIMTEXT)");
        assertEquals(MeineChiffre.BEISPIEL_GEHEIMTEXT, MeineChiffre.verschluesseln(MeineChiffre.BEISPIEL_KLARTEXT, MeineChiffre.BEISPIEL_SCHLUESSEL), "Dein Programm liefert nicht den Geheimtext aus dem Baukasten - Reihenfolge der Stufen und Schluesselformat pruefen");
    }

    @Test
    public void umkehrbar_kurz() {
        String k = MeineChiffre.TESTSCHLUESSEL_1;
        String g = MeineChiffre.verschluesseln("Ebbe und Flut", k);
        assertEquals("EBBEUNDFLUT", MeineChiffre.entschluesseln(g, k), "entschluesseln(verschluesseln(t)) muss den normalisierten Klartext liefern - Stufen in UMGEKEHRTER Reihenfolge zurueckrechnen (Falle P3)");
    }

    @Test
    public void umkehrbar_lang() {
        String t = "Im Maschinenzeitalter ersetzten Walzen und Lochstreifen die Feder. Wer heute eine Chiffre entwirft, muss davon ausgehen, dass der Gegner das Verfahren kennt. Geheim bleibt allein der Schluessel.";
        String k = MeineChiffre.TESTSCHLUESSEL_2;
        String g = MeineChiffre.verschluesseln(t, k);
        assertEquals(Hilfe.normalisiere(t), MeineChiffre.entschluesseln(g, k), "Langer Text (ueber 150 Buchstaben) wird nicht korrekt zurueckgewonnen - unvollstaendige Bloecke/Spalten pruefen (Falle P4)");
    }

    @Test
    public void umkehrbar_randfaelle() {
        String k = MeineChiffre.TESTSCHLUESSEL_1;
        assertEquals("A", MeineChiffre.entschluesseln(MeineChiffre.verschluesseln("a", k), k), "Ein einzelner Buchstabe muss funktionieren");
        assertEquals("QUIZ", MeineChiffre.entschluesseln(MeineChiffre.verschluesseln("Quiz!", k), k), "\"Quiz!\" -> nach Normalisierung \"QUIZ\"");
    }

    @Test
    public void geheimtext_nurErlaubteZeichen() {
        String g = MeineChiffre.verschluesseln("Ebbe und Flut", MeineChiffre.TESTSCHLUESSEL_1);
        assertTrue(g.length() > 0, "verschluesseln liefert noch einen leeren Text");
        assertTrue(nurBuchstabenOderHex(g), "Der Geheimtext darf nur A-Z bzw. bei XOR die Hexziffern 0-9, A-F enthalten: " + g);
    }

    @Test
    public void schluessel_wirkt() {
        String t = "Ebbe und Flut";
        String g1 = MeineChiffre.verschluesseln(t, MeineChiffre.TESTSCHLUESSEL_1);
        String g2 = MeineChiffre.verschluesseln(t, MeineChiffre.TESTSCHLUESSEL_2);
        assertFalse(g1.equals(g2), "Zwei verschiedene Schluessel liefern denselben Geheimtext - wirkt dein Schluessel ueberhaupt?");
        assertFalse(g1.equals(Hilfe.normalisiere(t)), "Der Geheimtext ist gleich dem Klartext - schwacher Schluessel (Falle P7)?");
    }
}
